import scrapy
import re

class EcomProductSpider(scrapy.Spider):
    name = "ecom_catalog"
    
    def parse(self, response):
        raw_sku = response.css('[data-product-id]::attr(data-product-id)').get()
        sku = re.search(r'^SKU-([0-9]+)', raw_sku.strip()).group(1)
        
        price_raw = response.css('.price-current::text').get()
        price = float(re.sub(r'[^\d.]', '', price_raw))
        
        yield {
            'sku': sku,
            'current_price': price,
            'colors': response.css('.swatch-list li::attr(data-color)').getall(),
            'breadcrumbs': ' > '.join(response.css('nav.breadcrumbs a::text').getall())
        }
